01项目概览

一句话定位与关键属性。

《第二绿洲》是一款面向 2~4 人 的 网页策略桌游。玩家分别扮演四个国家，围绕月球（公共领土）展开领土争夺。游戏以掷骰为核心随机源，融合 稳健 / 激进 策略选择、连击开发 的风险博弈，以及由 抢夺 升级而来的 崩坏 熔断机制。当公共领土被瓜分殆尽，私有领土最多者获胜。

掷骰驱动风险 / 收益博弈公共池衰减 零和守恒回合制像素风太空殖民

背景与主题

地球资源枯竭后，人类将目光投向月球——代号“第二绿洲”。四个国家率先具备登月能力，意图在月球表面抢占尽可能多的领土。然而月球公共领土有限，争夺必然走向零和：当最后一块公共领土被分配完毕，谁在私人物业上积累最多，谁就掌握了新文明的主动权。基调：冷峻的太空殖民竞争 + 轻策略博弈；视觉上采用 深空蓝绿 #0e7c86 + 月壤灰配色。

项目定位与优先级

  • 性质：作品集展示项目，需交付完整可玩产品。
  • 开发者：单人独立开发，WorkBuddy 全程参与（写代码 / 设计数值 / 写文档 / 审查机制）。
  • 开发优先级：A 核心规则逻辑 → B 界面好看 → C 联机对战 → D AI 对手 → E 音效动画（当前五大优先级全部完成）。

抢夺（§10）

当占领使公共 < 0（M > 公共剩余，Q1 裁定 m = 0）时，溢出部分 m2 触发抢夺，由发起者 (p) 发起。仅本局首次溢出触发；其后升级为崩坏。

// C3 方案 E：低者私有可能不足 m2，优先严格守恒 LowPrivate = 低者私有原值 Transfer = min(m2, LowPrivate) // 实际可转移量 r = min(r, Transfer) // r 受 Transfer 约束 低者私有' = 低者私有 − Transfer 高者私有' = 高者私有 + (Transfer − r) 公共' = 公共 + r // r 回归公共池 // 守恒校验：−Transfer + (Transfer − r) + r = 0
要点防守者 = 除发起者外当前私有最高者（并列取 ID 最小，确定性便于联机复现）。Q5：平手双方都重掷直至分高低。C6：抢夺裁决用单骰 1~6；加赛用双骰 2~12。

崩坏（§11）

// Q6 裁定：总量守恒目标 = x · m2 TotalTarget = x × m2 PerCap = floor((x × m2) / 4) // 每位非发起者的随机损失上限 RandomLoss_i = NextInt(0, PerCap) // 非发起者独立随机损失（C4 clamp） InitiatorLoss = TotalTarget − ΣOthers // 发起者承担剩余缺口 Public' = max(0, 公共 − x) // C5：clamp 至 0，=0 立即终局 NextX = x + 1 // 系数自增

发射（§12）与终局（§3）

  • 发射：首轮专属序章，掷双骰 ≥7 成功 +2；全员成功才进主循环；开发过度后需重新发射。
  • 终局：公共领土归零（=0）即终局，私有最高者胜；Q7 平局进入加赛：仅平手者掷双骰（2~12），高者胜，仍平手继续。
  • 荒地边界：若终局全员荒地（私有=0），则全员并列，仍进加赛决出唯一胜者。

04核心数据流：占领结算公式（§7.2）

实际占领量 M = RawGain × Multiplier（Multiplier 来自开发链；倒扣时 RawGain 为负，M 也为负）。

// 情形 A：M ≤ 0（倒扣 / 不开发）→ 私有减少、公共回血，守恒，不溢出 Private' = max(0, Private − |M|) // 私有 clamp 至 0 Public' = Public + (Private − Private') // 公共只加实际扣量，严格守恒 // 情形 B：M > 0 且 M ≤ 公共剩余 → 正常占领 Public' = Public − M Private' = Private + M // 情形 C：M > 公共剩余 → 溢出（C1） m2 = M − Public // 溢出的是实际占领量 M Public' = 0 Private' = Private + Public // 仅拿到公共剩余部分 IsOverflow = true // 剩余 m2 进入抢夺 / 崩坏
举例公共剩 30，玩家激进掷 (6,6)=12 且第 1 次连续对子（×2）→ M = 12 × 2 = 24。若公共剩 10，则 m2 = 24 − 10 = 14 进入抢夺。×2 倍率会显著放大溢出概率。

DiceRoller — 掷骰（含倒扣、None 不耗随机源）

src/Core/DiceRoller.ts
private RollAggressive(): DiceRollResult {
  const D1 = this._Random.NextDie();
  const D2 = this._Random.NextDie();
  const Sum = D1 + D2;
  const IsDouble = D1 === D2;
  const IsDeducted = Sum <= 6;             // 激进 ≤6 倒扣
  const RawGain = IsDeducted ? -Sum : Sum;
  return { Mode: DiceMode.Aggressive, Dice: [D1, D2], Sum,
           IsDouble, IsDeducted, RawGain };
}
private RollNone(): DiceRollResult {       // 不开发：不消耗随机源
  return { Mode: DiceMode.None, Dice: [], Sum: 0,
           IsDouble: false, IsDeducted: false, RawGain: 0 };
}

DevChain — 开发链状态机（Q4/Q8/C2）

src/Core/DevChain.ts Advance()
Advance(IsDouble: boolean, Mode: DiceMode): DevChainOutcome {
  if (Mode === DiceMode.None) { // Q8：不开发 = 清零连击
    this._ConsecutiveDoubles = 0;
    return { Multiplier: DevMultiplier.None, IsOverload: false, NewConsecutiveDoubles: 0 };
  }
  if (!IsDouble) { // 非对子（含 Steady 单骰）→ 清零
    this._ConsecutiveDoubles = 0; return { Multiplier: DevMultiplier.None, ... };
  }
  this._ConsecutiveDoubles += 1;
  if (this._ConsecutiveDoubles >= DEV_CHAIN_OVERLOAD_THRESHOLD) { // 第3次 → 过度
    this._ConsecutiveDoubles = 0;
    return { Multiplier: DevMultiplier.None, IsOverload: true, NewConsecutiveDoubles: 0 };
  }
  if (this._ConsecutiveDoubles === 1) return { Multiplier: DevMultiplier.Dev, ... };   // ×2
  return { Multiplier: DevMultiplier.BigDev, ... };                                       // ×3
}

Occupation — 占领结算（纯计算，含倒扣守恒、溢出 m2）

src/Core/Occupation.ts Calculate()
Calculate(PublicBefore, PrivateBefore, RawGain, Multiplier, SprintBonus=0) {
  const AdjustedRawGain = RawGain > 0 ? RawGain + SprintBonus : RawGain;
  const M = AdjustedRawGain * Multiplier;     // M = RawGain × 倍率（Q4）
  if (M <= 0) return { ...CalculateDeduction(PublicBefore, PrivateBefore, M) };
  return { ...CalculateClaim(PublicBefore, M) };
}
// 正常占领：M ≤ 公共 → 公共−M，私有+M，不溢出
// 溢出：m2 = M − PublicBefore；公共=0，私有+PublicBefore，IsOverflow=true

export class GameStore extends EventEmitter<StoreEvents> implements IGameStore {
  PlayTurn(Mode) {
    const Result = this._State.PlayTurn(Mode);
    this._EmitPhase(); this._EmitRound();
    this.Emit('Turn', { Result, Snapshot });
    this.Emit('Snapshot', this._State.Snapshot);

  }

G-001 公敌税
